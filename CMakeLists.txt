cmake_minimum_required(VERSION 3.20)
project(indextts25_metal LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
find_path(PCRE2_INCLUDE_DIR pcre2.h HINTS /opt/homebrew/include REQUIRED)
find_library(PCRE2_LIBRARY pcre2-8 HINTS /opt/homebrew/lib REQUIRED)
find_path(SNDFILE_INCLUDE_DIR sndfile.h HINTS /opt/homebrew/include REQUIRED)
find_library(SNDFILE_LIBRARY sndfile HINTS /opt/homebrew/lib REQUIRED)
find_path(SOXR_INCLUDE_DIR soxr.h HINTS /opt/homebrew/include REQUIRED)
find_library(SOXR_LIBRARY soxr HINTS /opt/homebrew/lib REQUIRED)
set(BASE ${CMAKE_CURRENT_SOURCE_DIR}/vendor/metal-runtime)
set(EMBED ${CMAKE_CURRENT_BINARY_DIR}/generated/metal_kernels_embedded.inc)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/generated)
add_custom_command(OUTPUT ${EMBED}
  COMMAND ${CMAKE_COMMAND} -DINPUT=${BASE}/metal/kernels.metal -DOUTPUT=${EMBED}
    -P ${BASE}/cmake/embed_metal.cmake
  DEPENDS ${BASE}/metal/kernels.metal ${BASE}/cmake/embed_metal.cmake)
add_library(itts25_native STATIC
  ${BASE}/bundle.cpp ${BASE}/metal_context.mm ${EMBED}
  src/weights.cpp src/codec.cpp src/codec_ops.mm src/gpt.cpp src/gpt_sampling.cpp src/pipeline.cpp src/server.cpp src/qwen.cpp
  src/acoustic.cpp src/acoustic_ops.cpp src/emotion.cpp src/emotion_ops.cpp
  src/speech_features.cpp src/relative_attention.mm
  src/campplus.cpp src/campplus_ops.cpp src/camp_convolution.mm)
set_source_files_properties(${BASE}/metal_context.mm src/codec_ops.mm src/relative_attention.mm src/camp_convolution.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
target_include_directories(itts25_native PUBLIC include ${BASE}/include PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
include(vendor/fst.cmake)
add_library(itts25_frontend STATIC src/tokenizer.cpp src/audio_frontend.cpp src/audio_decode.mm src/unicode.mm src/text_processor.cpp)
set_source_files_properties(src/unicode.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
set_source_files_properties(src/text_processor.cpp PROPERTIES COMPILE_OPTIONS "-w")
target_include_directories(itts25_frontend PUBLIC include vendor/headers ${BASE}/include PRIVATE ${PCRE2_INCLUDE_DIR} ${SNDFILE_INCLUDE_DIR} ${SOXR_INCLUDE_DIR})
target_link_libraries(itts25_frontend PUBLIC itts25_native itts25_kaldifst ${PCRE2_LIBRARY} ${SNDFILE_LIBRARY} ${SOXR_LIBRARY} "-framework AudioToolbox")
target_compile_options(itts25_frontend PRIVATE -Wall -Wextra)
add_executable(itts25-frontend src/frontend_main.cpp)
target_link_libraries(itts25-frontend PRIVATE itts25_frontend)
add_library(itts25_runtime STATIC src/native_voice.cpp src/native_runtime.cpp)
target_link_libraries(itts25_runtime PUBLIC itts25_frontend)
target_compile_options(itts25_runtime PRIVATE -Wall -Wextra)
add_executable(itts25-native src/native_main.cpp src/http.cpp vendor/headers/httplib.cpp)
target_link_libraries(itts25-native PRIVATE itts25_runtime sqlite3)
target_link_libraries(itts25_native PUBLIC "-framework Foundation" "-framework Metal" "-framework MetalPerformanceShaders")
target_compile_options(itts25_native PRIVATE -Wall -Wextra)
add_executable(itts25-metal src/main.cpp)
target_link_libraries(itts25-metal PRIVATE itts25_native)
enable_testing()
add_test(NAME metal_device COMMAND itts25-metal --diagnostics)
add_test(NAME mps_zero_bias_reuse COMMAND itts25-metal --test-mps-zero-bias)
set(ITTS25_MODEL_BUNDLE "" CACHE PATH "Real 2.5 weight bundle")
set(ITTS25_CODEC_GOLDEN "" CACHE PATH "Codec golden fixture bundle")
set(ITTS25_GPT_MODEL "" CACHE PATH "Full 2.5 model bundle")
set(ITTS25_GPT_GOLDEN "" CACHE PATH "GPT golden fixture bundle")
set(ITTS25_ACOUSTIC_GOLDEN "" CACHE PATH "Acoustic golden fixture bundle")
set(ITTS25_EMOTION_GOLDEN "" CACHE PATH "Emotion golden fixture bundle")
set(ITTS25_SPEECH_GOLDEN "" CACHE PATH "W2V-BERT golden fixture bundle")
set(ITTS25_CAMP_GOLDEN "" CACHE PATH "CAMPPlus golden fixture bundle")
if(ITTS25_MODEL_BUNDLE AND ITTS25_CODEC_GOLDEN)
  add_test(NAME codec_parity COMMAND itts25-metal --test-codec ${ITTS25_MODEL_BUNDLE} ${ITTS25_CODEC_GOLDEN})
  set_tests_properties(codec_parity PROPERTIES TIMEOUT 240)
endif()
if(ITTS25_GPT_MODEL AND ITTS25_ACOUSTIC_GOLDEN)
  add_test(NAME acoustic_parity COMMAND itts25-metal --test-acoustic ${ITTS25_GPT_MODEL} ${ITTS25_ACOUSTIC_GOLDEN})
  set_tests_properties(acoustic_parity PROPERTIES TIMEOUT 240)
endif()
if(ITTS25_GPT_MODEL AND ITTS25_GPT_GOLDEN)
  add_test(NAME gpt_parity COMMAND itts25-metal --test-gpt ${ITTS25_GPT_MODEL} ${ITTS25_GPT_GOLDEN})
  set_tests_properties(gpt_parity PROPERTIES TIMEOUT 240)
endif()
if(ITTS25_GPT_MODEL AND ITTS25_EMOTION_GOLDEN)
  add_test(NAME emotion_parity COMMAND itts25-metal --test-emotion ${ITTS25_GPT_MODEL} ${ITTS25_EMOTION_GOLDEN})
  set_tests_properties(emotion_parity PROPERTIES TIMEOUT 240)
endif()
if(ITTS25_GPT_MODEL AND ITTS25_SPEECH_GOLDEN)
  add_test(NAME speech_parity COMMAND itts25-metal --test-speech ${ITTS25_GPT_MODEL} ${ITTS25_SPEECH_GOLDEN})
  set_tests_properties(speech_parity PROPERTIES TIMEOUT 240)
endif()
if(ITTS25_GPT_MODEL AND ITTS25_CAMP_GOLDEN)
  add_test(NAME camp_parity COMMAND itts25-metal --test-camp ${ITTS25_GPT_MODEL} ${ITTS25_CAMP_GOLDEN})
  set_tests_properties(camp_parity PROPERTIES TIMEOUT 240)
endif()

set(ITTS25_QWEN_GOLDEN "" CACHE PATH "Qwen3 golden fixture bundle")
if(ITTS25_GPT_MODEL AND ITTS25_QWEN_GOLDEN)
  add_test(NAME qwen_parity COMMAND itts25-metal --test-qwen ${ITTS25_GPT_MODEL} ${ITTS25_QWEN_GOLDEN})
  set_tests_properties(qwen_parity PROPERTIES TIMEOUT 240)
endif()

set(ITTS25_SAMPLING_GOLDEN "" CACHE PATH "Sampling processor fixtures")
if(ITTS25_SAMPLING_GOLDEN)
  add_test(NAME sampling_parity COMMAND itts25-metal --test-sampling ${ITTS25_SAMPLING_GOLDEN})
endif()

# Explicit benchmark target: repeated TTS in one process, model construction timed separately.
add_executable(itts25-benchmark src/benchmark_main.cpp)
target_link_libraries(itts25-benchmark PRIVATE itts25_runtime)
